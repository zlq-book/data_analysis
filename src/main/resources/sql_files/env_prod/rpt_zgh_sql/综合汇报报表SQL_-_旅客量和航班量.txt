-------------------------
--   旅客量和航班量    --
-------------------------
SELECT   -- 表头不要动，表头不要动，表头不要动，重要的事情说三遍!!!
CUR.FST_INDICATOR,   -- 一级指标
CUR.SEC_INDICATOR,   -- 二级指标
CUR.CUR_PERIOD,     -- 当期运输
MOM.MOM_PERIOD,      -- 环期运输
YOY.YOY_PERIOD,      -- 同期运输
CUR_MON.CUR_MON_PERIOD, -- 当月当期运输
MOM_MON.MOM_MON_PERIOD, -- 当月环期运输
YOY_MON.YOY_MON_PERIOD, -- 当月同期运输
CUR_YEAR.CUR_YEAR_PERIOD, -- 当年当期运输
YOY_YEAR.YOY_YEAR_PERIOD  -- 当年同期运输
FROM 
(
	SELECT 
		'旅客运输量(万人)'  AS FST_INDICATOR,
		'' AS SEC_INDICATOR, 
		COUNT(SEG.FK_PASSENGER_USER_TID) AS CUR_PERIOD  -- 当期运输						 
	FROM T_DWD_DEPART_SEG_FACT SEG	
	WHERE 1=1
	<START_DATE>
	AND SEG.FK_DEPARTURES_DATE >= ?{START_DATE}
	</START_DATE>
	<END_DATE>
	AND SEG.FK_DEPARTURES_DATE <= ?{END_DATE}
	</END_DATE>   
) CUR
LEFT JOIN
(
	SELECT 
		'旅客运输量(万人)'  AS FST_INDICATOR,
		'' AS SEC_INDICATOR,
		COUNT(SEG.FK_PASSENGER_USER_TID) AS MOM_PERIOD  -- 环期运输
	FROM T_DWD_DEPART_SEG_FACT SEG
	WHERE 1=1 
	<START_DATE>
	AND SEG.FK_DEPARTURES_DATE >= DATE_SUB(CAST(?{START_DATE} AS DATE), 
	</START_DATE>
	<END_DATE>
	INTERVAL (DATEDIFF(CAST(?{END_DATE} AS DATE),
	</END_DATE>
	<START_DATE>
	CAST(?{START_DATE} AS DATE))+1) DAY)
	AND SEG.FK_DEPARTURES_DATE <= DATE_SUB(CAST(?{START_DATE} AS DATE), INTERVAL 1 DAY)
	</START_DATE> 
) MOM ON CUR.FST_INDICATOR = MOM.FST_INDICATOR AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(MOM.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'旅客运输量(万人)'  AS FST_INDICATOR,
		'' AS SEC_INDICATOR,
		COUNT(SEG.FK_PASSENGER_USER_TID) AS YOY_PERIOD   -- 同期运输
	FROM T_DWD_DEPART_SEG_FACT SEG
	WHERE 1=1 
	<START_DATE>
	AND SEG.FK_DEPARTURES_DATE >= DATE_SUB(CAST(?{START_DATE} AS DATE), INTERVAL 1 YEAR)
	</START_DATE>
	<END_DATE>
	AND SEG.FK_DEPARTURES_DATE <= DATE_SUB(CAST(?{END_DATE} AS DATE), INTERVAL 1 YEAR)
	</END_DATE> 
) YOY ON CUR.FST_INDICATOR = YOY.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(YOY.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'旅客运输量(万人)'  AS FST_INDICATOR, 
		'' AS SEC_INDICATOR, 
		COUNT(SEG.FK_PASSENGER_USER_TID) AS CUR_MON_PERIOD  -- 当月当期运输
	FROM T_DWD_DEPART_SEG_FACT SEG
	WHERE 1=1 
	<END_DATE>
	AND SEG.FK_DEPARTURES_DATE >= DATE_FORMAT(CAST(?{END_DATE} AS DATE), '%Y-%m-01') 
	AND SEG.FK_DEPARTURES_DATE <= ?{END_DATE}
	</END_DATE> 
) CUR_MON ON CUR.FST_INDICATOR = CUR_MON.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(CUR_MON.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'旅客运输量(万人)'  AS FST_INDICATOR, 
		'' AS SEC_INDICATOR, 
		COUNT(SEG.FK_PASSENGER_USER_TID) AS MOM_MON_PERIOD  -- 当月环期运输
	FROM T_DWD_DEPART_SEG_FACT SEG
	WHERE 1=1 
	-- END_DATE 当月1日
	<END_DATE>
	AND SEG.FK_DEPARTURES_DATE >= DATE_SUB(DATE_FORMAT(CAST(?{END_DATE} AS DATE), '%Y-%m-01'), INTERVAL 1 MONTH)
	AND SEG.FK_DEPARTURES_DATE <= DATE_SUB(CAST(?{END_DATE} AS DATE), INTERVAL 1 MONTH)
	</END_DATE> 
) MOM_MON ON CUR.FST_INDICATOR = MOM_MON.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(MOM_MON.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'旅客运输量(万人)'  AS FST_INDICATOR, 
		'' AS SEC_INDICATOR, 
		COUNT(SEG.FK_PASSENGER_USER_TID) AS YOY_MON_PERIOD  -- 当月同期运输
	FROM T_DWD_DEPART_SEG_FACT SEG
	WHERE 1=1 
	<END_DATE>
	AND SEG.FK_DEPARTURES_DATE >= DATE_SUB(DATE_FORMAT(CAST(?{END_DATE} AS DATE), '%Y-%m-01'), INTERVAL 1 YEAR)
	AND SEG.FK_DEPARTURES_DATE <= DATE_SUB(CAST(?{END_DATE} AS DATE), INTERVAL 1 YEAR)
	</END_DATE> 
) YOY_MON ON CUR.FST_INDICATOR = YOY_MON.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(YOY_MON.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'旅客运输量(万人)'  AS FST_INDICATOR, 
		'' AS SEC_INDICATOR, 
		COUNT(SEG.FK_PASSENGER_USER_TID) AS CUR_YEAR_PERIOD  -- 当年当期运输
	FROM T_DWD_DEPART_SEG_FACT SEG
	WHERE 1=1 
	<END_DATE>
	AND SEG.FK_DEPARTURES_DATE >= DATE_FORMAT(CAST(?{END_DATE} AS DATE), '%Y-01-01')
	AND SEG.FK_DEPARTURES_DATE <= ?{END_DATE}
	</END_DATE> 
) CUR_YEAR ON CUR.FST_INDICATOR = CUR_YEAR.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(CUR_YEAR.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'旅客运输量(万人)'  AS FST_INDICATOR, 
		'' AS SEC_INDICATOR, 
		COUNT(SEG.FK_PASSENGER_USER_TID) AS YOY_YEAR_PERIOD  -- 当年同期运输
	FROM T_DWD_DEPART_SEG_FACT SEG
	WHERE 1=1 
	<END_DATE>
	AND SEG.FK_DEPARTURES_DATE >= DATE_SUB(DATE_FORMAT(CAST(?{END_DATE} AS DATE), '%Y-01-01'), INTERVAL 1 YEAR)
	AND SEG.FK_DEPARTURES_DATE <= DATE_SUB(CAST(?{END_DATE} AS DATE), INTERVAL 1 YEAR)
	</END_DATE> 
) YOY_YEAR ON CUR.FST_INDICATOR = YOY_YEAR.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(YOY_YEAR.SEC_INDICATOR,'')
UNION ALL

(
WITH SEG_DIM AS (
	SELECT  DISTINCT
		SEG_TMP.OPERAT_AIRLINE,SEG_TMP.OPERAT_FLIGHT_NUM,SEG_TMP.DEPARTURE_DATE,
		1 AS FLIGHT_COUNT					 
	FROM DIM_PROD.T_DIM_SEG_DIM SEG_TMP
)
SELECT   -- 表头不要动，表头不要动，表头不要动，重要的事情说三遍!!!
CUR.FST_INDICATOR,   -- 一级指标

(
	SELECT 
		'公司航班量'  AS FST_INDICATOR,

(
	SELECT 
		'公司航班量'  AS FST_INDICATOR, 
		'' AS SEC_INDICATOR, 
		SUM(SEG_DIM.FLIGHT_COUNT) AS CUR_MON_PERIOD  -- 当月当期运输
