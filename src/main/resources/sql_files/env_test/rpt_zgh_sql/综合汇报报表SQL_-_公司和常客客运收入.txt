--------------------------
--   公司客运收入(万元) --
--------------------------
SELECT   -- 表头不要动，表头不要动，表头不要动，重要的事情说三遍!!!
CUR.FST_INDICATOR,   -- 一级指标
CUR.SEC_INDICATOR,   -- 二级指标
CUR.CUR_PERIOD_TKT,     -- 当期销售
CUR.CUR_PERIOD_DPT,     -- 当期运输
MOM.MOM_PERIOD_TKT,      -- 环期销售
MOM.MOM_PERIOD_DPT,      -- 环期运输
YOY.YOY_PERIOD_TKT,      -- 同期销售
YOY.YOY_PERIOD_DPT,      -- 同期运输
CUR_MON.CUR_MON_PERIOD_TKT, -- 当月当期销售
CUR_MON.CUR_MON_PERIOD_DPT, -- 当月当期运输
MOM_MON.MOM_MON_PERIOD_TKT, -- 当月环期销售
MOM_MON.MOM_MON_PERIOD_DPT, -- 当月环期运输
YOY_MON.YOY_MON_PERIOD_TKT, -- 当月同期销售
YOY_MON.YOY_MON_PERIOD_DPT, -- 当月同期运输
CUR_YEAR.CUR_YEAR_PERIOD_TKT, -- 当年当期销售
CUR_YEAR.CUR_YEAR_PERIOD_DPT, -- 当年当期运输
YOY_YEAR.YOY_YEAR_PERIOD_TKT, -- 当年同期销售
YOY_YEAR.YOY_YEAR_PERIOD_DPT  -- 当年同期运输
FROM 
(
	SELECT 
		'公司客运收入(万元)'  AS FST_INDICATOR,
		'' AS SEC_INDICATOR, 
		SUM(SEG.TKT_AK_SALE_PRICE_CURRENCY) AS CUR_PERIOD_TKT,  -- 当期销售
		SUM(SEG.DPT_SEG_TRANSPORT_TIK_PRICE) AS CUR_PERIOD_DPT  -- 当期运输						 
	FROM FFP_REPORT_TABLE_VIEW SEG	
	WHERE 1=1
	<START_DATE>
	AND SEG.DPT_FK_DEPARTURES_DATE >= ?{START_DATE}
	</START_DATE>
	<END_DATE>
	AND SEG.DPT_FK_DEPARTURES_DATE <= ?{END_DATE}
	</END_DATE>   
) CUR
LEFT JOIN
(
	SELECT 
		'公司客运收入(万元)'  AS FST_INDICATOR,
		'' AS SEC_INDICATOR,
		SUM(SEG.TKT_AK_SALE_PRICE_CURRENCY) AS MOM_PERIOD_TKT,  -- 环期销售
		SUM(SEG.DPT_SEG_TRANSPORT_TIK_PRICE) AS MOM_PERIOD_DPT  -- 环期运输
	FROM FFP_REPORT_TABLE_VIEW SEG
	WHERE 1=1 
	<START_DATE>
	AND SEG.DPT_FK_DEPARTURES_DATE >= DATE_SUB(CAST(?{START_DATE} AS DATE), 
	</START_DATE>
	<END_DATE>
	INTERVAL (DATEDIFF(CAST(?{END_DATE} AS DATE),
	</END_DATE>
	<START_DATE>
	CAST(?{START_DATE} AS DATE))+1) DAY)
	AND SEG.DPT_FK_DEPARTURES_DATE <= DATE_SUB(CAST(?{START_DATE} AS DATE), INTERVAL 1 DAY)
	</START_DATE> 
) MOM ON CUR.FST_INDICATOR = MOM.FST_INDICATOR AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(MOM.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'公司客运收入(万元)'  AS FST_INDICATOR,
		'' AS SEC_INDICATOR,
		SUM(SEG.TKT_AK_SALE_PRICE_CURRENCY) AS YOY_PERIOD_TKT,	 -- 同期销售
		SUM(SEG.DPT_SEG_TRANSPORT_TIK_PRICE) AS YOY_PERIOD_DPT   -- 同期运输
	FROM FFP_REPORT_TABLE_VIEW SEG
	WHERE 1=1 
	<START_DATE>
	AND SEG.DPT_FK_DEPARTURES_DATE >= DATE_SUB(CAST(?{START_DATE} AS DATE), INTERVAL 1 YEAR)
	</START_DATE>
	<END_DATE>
	AND SEG.DPT_FK_DEPARTURES_DATE <= DATE_SUB(CAST(?{END_DATE} AS DATE), INTERVAL 1 YEAR)
	</END_DATE> 
) YOY ON CUR.FST_INDICATOR = YOY.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(YOY.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'公司客运收入(万元)'  AS FST_INDICATOR, 
		'' AS SEC_INDICATOR, 
		SUM(SEG.TKT_AK_SALE_PRICE_CURRENCY) AS CUR_MON_PERIOD_TKT,	-- 当月当期销售
		SUM(SEG.DPT_SEG_TRANSPORT_TIK_PRICE) AS CUR_MON_PERIOD_DPT  -- 当月当期运输
	FROM FFP_REPORT_TABLE_VIEW SEG
	WHERE 1=1 
	<END_DATE>
	AND SEG.DPT_FK_DEPARTURES_DATE >= DATE_FORMAT(CAST(?{END_DATE} AS DATE), '%Y-%m-01') 
	AND SEG.DPT_FK_DEPARTURES_DATE <= ?{END_DATE}
	</END_DATE> 
) CUR_MON ON CUR.FST_INDICATOR = CUR_MON.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(CUR_MON.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'公司客运收入(万元)'  AS FST_INDICATOR, 
		'' AS SEC_INDICATOR, 
		SUM(SEG.TKT_AK_SALE_PRICE_CURRENCY) AS MOM_MON_PERIOD_TKT,	-- 当月环期销售
		SUM(SEG.DPT_SEG_TRANSPORT_TIK_PRICE) AS MOM_MON_PERIOD_DPT  -- 当月环期运输
	FROM FFP_REPORT_TABLE_VIEW SEG
	WHERE 1=1 
	-- END_DATE 当月1日
	<END_DATE>
	AND SEG.DPT_FK_DEPARTURES_DATE >= DATE_SUB(DATE_FORMAT(CAST(?{END_DATE} AS DATE), '%Y-%m-01'), INTERVAL 1 MONTH)
	AND SEG.DPT_FK_DEPARTURES_DATE <= DATE_SUB(CAST(?{END_DATE} AS DATE), INTERVAL 1 MONTH)
	</END_DATE> 
) MOM_MON ON CUR.FST_INDICATOR = MOM_MON.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(MOM_MON.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'公司客运收入(万元)'  AS FST_INDICATOR, 
		'' AS SEC_INDICATOR, 
		SUM(SEG.TKT_AK_SALE_PRICE_CURRENCY) AS YOY_MON_PERIOD_TKT,	-- 当月同期销售
		SUM(SEG.DPT_SEG_TRANSPORT_TIK_PRICE) AS YOY_MON_PERIOD_DPT  -- 当月同期运输
	FROM FFP_REPORT_TABLE_VIEW SEG
	WHERE 1=1 
	<END_DATE>
	AND SEG.DPT_FK_DEPARTURES_DATE >= DATE_SUB(DATE_FORMAT(CAST(?{END_DATE} AS DATE), '%Y-%m-01'), INTERVAL 1 YEAR)
	AND SEG.DPT_FK_DEPARTURES_DATE <= DATE_SUB(CAST(?{END_DATE} AS DATE), INTERVAL 1 YEAR)
	</END_DATE> 
) YOY_MON ON CUR.FST_INDICATOR = YOY_MON.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(YOY_MON.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'公司客运收入(万元)'  AS FST_INDICATOR, 
		'' AS SEC_INDICATOR, 
		SUM(SEG.TKT_AK_SALE_PRICE_CURRENCY) AS CUR_YEAR_PERIOD_TKT,	 -- 当年当期销售
		SUM(SEG.DPT_SEG_TRANSPORT_TIK_PRICE) AS CUR_YEAR_PERIOD_DPT  -- 当年当期运输
	FROM FFP_REPORT_TABLE_VIEW SEG
	WHERE 1=1 
	<END_DATE>
	AND SEG.DPT_FK_DEPARTURES_DATE >= DATE_FORMAT(CAST(?{END_DATE} AS DATE), '%Y-01-01')
	AND SEG.DPT_FK_DEPARTURES_DATE <= ?{END_DATE}
	</END_DATE> 
) CUR_YEAR ON CUR.FST_INDICATOR = CUR_YEAR.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(CUR_YEAR.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'公司客运收入(万元)'  AS FST_INDICATOR, 
		'' AS SEC_INDICATOR, 
		SUM(SEG.TKT_AK_SALE_PRICE_CURRENCY) AS YOY_YEAR_PERIOD_TKT,	 -- 当年同期销售
		SUM(SEG.DPT_SEG_TRANSPORT_TIK_PRICE) AS YOY_YEAR_PERIOD_DPT  -- 当年同期运输
	FROM FFP_REPORT_TABLE_VIEW SEG
	WHERE 1=1 
	<END_DATE>
	AND SEG.DPT_FK_DEPARTURES_DATE >= DATE_SUB(DATE_FORMAT(CAST(?{END_DATE} AS DATE), '%Y-01-01'), INTERVAL 1 YEAR)
	AND SEG.DPT_FK_DEPARTURES_DATE <= DATE_SUB(CAST(?{END_DATE} AS DATE), INTERVAL 1 YEAR)
	</END_DATE> 
) YOY_YEAR ON CUR.FST_INDICATOR = YOY_YEAR.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(YOY_YEAR.SEC_INDICATOR,'')
UNION ALL
