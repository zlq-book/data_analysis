WITH MEM_VIP AS (
	SELECT DATE,NAME,'银卡' AS LEVEL,SILVER AS AMOUNT FROM T_DWD_CLK_VIP_MEMBER_TOTAL_FACT 
	UNION ALL
	SELECT DATE,NAME,'金卡' AS LEVEL,GOLD AS AMOUNT FROM T_DWD_CLK_VIP_MEMBER_TOTAL_FACT 
	UNION ALL
	SELECT DATE,NAME,'白金卡' AS LEVEL,PLATINUM AS AMOUNT FROM T_DWD_CLK_VIP_MEMBER_TOTAL_FACT
	UNION ALL
	SELECT DATE,NAME,'终身白金卡' AS LEVEL,FINAL_WHITE AS AMOUNT FROM T_DWD_CLK_VIP_MEMBER_TOTAL_FACT
	UNION ALL
	SELECT DATE,NAME,'贵宾会员' AS LEVEL,TOTAL_BY_ATTRIBUTES AS AMOUNT FROM T_DWD_CLK_VIP_MEMBER_TOTAL_FACT
)
SELECT   -- 表头不要动，表头不要动，表头不要动，重要的事情说三遍!!!
CUR.FST_INDICATOR,   -- 一级指标
CUR.SEC_INDICATOR, -- 二级指标
CUR.CUR_PERIOD ,     -- 当期
MOM.MOM_PERIOD,      -- 环期
YOY.YOY_PERIOD,      -- 同期
CUR_MON.CUR_MON_PERIOD, -- 当月当期
MOM_MON.MOM_MON_PERIOD, -- 当月环期
YOY_MON.YOY_MON_PERIOD, -- 当月同期
CUR_YEAR.CUR_YEAR_PERIOD, -- 当年当期
YOY_YEAR.YOY_YEAR_PERIOD  -- 当年同期
FROM 
(
	SELECT 
		'会员量' AS FST_INDICATOR,
		MEM_VIP.LEVEL  AS SEC_INDICATOR,
		SUM(MEM_VIP.AMOUNT) AS CUR_PERIOD -- 当期	
	FROM MEM_VIP	
	WHERE 1=1 
	AND MEM_VIP.NAME = 'SC'
	<END_DATE>
	AND MEM_VIP.DATE = ?{END_DATE}
	</END_DATE>
	GROUP BY MEM_VIP.LEVEL  
) CUR
LEFT JOIN
(
	SELECT 
		'会员量' AS FST_INDICATOR, 
		MEM_VIP.LEVEL  AS SEC_INDICATOR,
		SUM(MEM_VIP.AMOUNT) AS MOM_PERIOD  -- 环期
	FROM MEM_VIP
	WHERE 1=1
	AND MEM_VIP.NAME = 'SC'
	<START_DATE>
	AND MEM_VIP.DATE = DATE_SUB(CAST(?{START_DATE} AS DATE), INTERVAL 1 DAY)
	</START_DATE>
	GROUP BY MEM_VIP.LEVEL
) MOM ON CUR.FST_INDICATOR = MOM.FST_INDICATOR AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(MOM.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'会员量' AS FST_INDICATOR, 
		MEM_VIP.LEVEL  AS SEC_INDICATOR,
		SUM(MEM_VIP.AMOUNT) AS YOY_PERIOD   -- 同期
	FROM MEM_VIP
	WHERE 1=1
	AND MEM_VIP.NAME = 'SC'
	<END_DATE>
	AND MEM_VIP.DATE = DATE_SUB(CAST(?{END_DATE} AS DATE), INTERVAL 1 YEAR)
	</END_DATE>
	GROUP BY MEM_VIP.LEVEL
) YOY ON CUR.FST_INDICATOR = YOY.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(YOY.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'会员量' AS FST_INDICATOR,  
		MEM_VIP.LEVEL  AS SEC_INDICATOR, 
		SUM(MEM_VIP.AMOUNT) AS CUR_MON_PERIOD  -- 当月当期
	FROM MEM_VIP
	WHERE 1=1
	AND MEM_VIP.NAME = 'SC'
	<END_DATE>
	AND MEM_VIP.DATE = ?{END_DATE}
	</END_DATE>
	GROUP BY MEM_VIP.LEVEL
) CUR_MON ON CUR.FST_INDICATOR = CUR_MON.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(CUR_MON.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'会员量' AS FST_INDICATOR,  
		MEM_VIP.LEVEL  AS SEC_INDICATOR, 
		SUM(MEM_VIP.AMOUNT) AS MOM_MON_PERIOD  -- 当月环期
	FROM MEM_VIP
	WHERE 1=1
	AND MEM_VIP.NAME = 'SC'
	<END_DATE>
	AND MEM_VIP.DATE = DATE_SUB(CAST(?{END_DATE} AS DATE), INTERVAL 1 MONTH)
	</END_DATE>
	GROUP BY MEM_VIP.LEVEL
) MOM_MON ON CUR.FST_INDICATOR = MOM_MON.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(MOM_MON.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'会员量' AS FST_INDICATOR,  
		MEM_VIP.LEVEL  AS SEC_INDICATOR, 
		SUM(MEM_VIP.AMOUNT) AS YOY_MON_PERIOD  -- 当月同期
	FROM MEM_VIP
	WHERE 1=1
	AND MEM_VIP.NAME = 'SC'
	<END_DATE>
	AND MEM_VIP.DATE = DATE_SUB(CAST(?{END_DATE} AS DATE), INTERVAL 1 YEAR)
	</END_DATE>
	GROUP BY MEM_VIP.LEVEL
) YOY_MON ON CUR.FST_INDICATOR = YOY_MON.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(YOY_MON.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'会员量' AS FST_INDICATOR,  
		MEM_VIP.LEVEL  AS SEC_INDICATOR, 
		SUM(MEM_VIP.AMOUNT) AS CUR_YEAR_PERIOD  -- 当年当期
	FROM MEM_VIP
	WHERE 1=1
	AND MEM_VIP.NAME = 'SC'
	<END_DATE>
	AND MEM_VIP.DATE = ?{END_DATE}
	</END_DATE>
	GROUP BY MEM_VIP.LEVEL
) CUR_YEAR ON CUR.FST_INDICATOR = CUR_YEAR.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(CUR_YEAR.SEC_INDICATOR,'')
LEFT JOIN
(
	SELECT 
		'会员量' AS FST_INDICATOR,  
		MEM_VIP.LEVEL  AS SEC_INDICATOR, 
		SUM(MEM_VIP.AMOUNT) AS YOY_YEAR_PERIOD  -- 当年同期
	FROM MEM_VIP
	WHERE 1=1
	AND MEM_VIP.NAME = 'SC'
	<END_DATE>
	AND MEM_VIP.DATE = DATE_SUB(CAST(?{END_DATE} AS DATE), INTERVAL 1 YEAR)
	</END_DATE>
	GROUP BY MEM_VIP.LEVEL
) YOY_YEAR ON CUR.FST_INDICATOR = YOY_YEAR.FST_INDICATOR  AND COALESCE(CUR.SEC_INDICATOR,'') = COALESCE(YOY_YEAR.SEC_INDICATOR,'')
UNION ALL
